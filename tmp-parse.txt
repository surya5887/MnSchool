const parseInlineBoxes = (html: string) => {
    return html.replace(/\{\{(.*?)\}\}/g, (match, p1) => {
        if (!p1.trim()) {
            return '<span style=\"display: inline-block; width: 60px; border-bottom: 1px dashed #000;\">&nbsp;</span>';
        }
        if (p1.includes('/')) {
            const opts = p1.split('/').map(o => o.trim()).join(' / ');
            return \[ \ ]\;
        }
        return \<span style=\"border: 1px solid #000; padding: 2px 8px; border-radius: 4px;\">\</span>\;
    });
};
